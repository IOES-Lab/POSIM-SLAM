cmake_minimum_required(VERSION 3.22)
project(posim_slam_native LANGUAGES CXX)
option(POSIM_SLAM_BUILD_ACOUSTIC "Build the OpenCV acoustic C ABI" ON)
option(POSIM_SLAM_BUILD_VISUAL "Build the Stella/ROS 2 adapter" OFF)
set(POSIM_SLAM_NATIVE_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../src/posim_slam/data/native)
if(POSIM_SLAM_BUILD_ACOUSTIC)
  find_package(OpenCV 4.13.0 EXACT REQUIRED COMPONENTS core imgproc video)
  add_library(wwos_acoustic_cv SHARED ${POSIM_SLAM_NATIVE_SOURCE}/acoustic_cv.cc)
  target_compile_features(wwos_acoustic_cv PRIVATE cxx_std_17)
  target_include_directories(wwos_acoustic_cv PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(wwos_acoustic_cv PRIVATE ${OpenCV_LIBS})
  install(TARGETS wwos_acoustic_cv LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
endif()
if(POSIM_SLAM_BUILD_VISUAL)
  add_subdirectory(${POSIM_SLAM_NATIVE_SOURCE}/visual visual)
endif()
